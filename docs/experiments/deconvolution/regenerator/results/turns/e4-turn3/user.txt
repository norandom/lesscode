### src/facdrone/shells/lifecycle_monitor_runner.py

```python
"""The lifecycle monitor runner after a served decision (facdrone-signal-lifecycle 10.2; R5.4
to R5.6, R5.9, R6.3, R7.3, R7.4, R8.5, R10.4, R12.1, R12.2, R12.4, R12.6 to R12.8). After
a decision's evidence is durable, the background runner evaluates
every scheduled window that is due for each monitored signal, appends the readings and any
qualified flag with the final window, committing each window independently, then emits
under the one standard with the severity the event declares: a flag is actionable, a watch
informational, a reading detail. A
window is due the day after its cutoff, so the evidence it is judged on is complete and a
re-measurement on demand reads the same window; a repeated serve at one cutoff creates no
window; a window without evidence is recorded not run with its reason. Readings are keyed
on the baseline they were judged against, so a re-admission under a new baseline starts
a fresh sequence. The runner receives a flag writer bound into the transaction and the
decision's identity, and a description of what is monitored; it never holds the store.
Nothing it does reaches the decision that was just served (R8.5): the committed flag
binds the next decision's standing through the one route. Every moment the runner
stamps is the injected ``as_of``, never the wall clock.
"""

from __future__ import annotations

import hashlib
import json
from collections.abc import Callable, Mapping, Sequence
from dataclasses import asdict, dataclass
from datetime import UTC, datetime, timedelta

from sqlalchemy import Connection, Engine, select
from sqlalchemy.dialects.postgresql import insert

from facdrone.pipelines.evaluation.monitor_windows import (
    MonitorDeclaration,
    Window,
    due_windows,
    persistence,
    scheduled_windows,
)
from facdrone.schema import SIGNAL_MONITOR_READING
from facdrone_kernel.domain.signal_lifecycle import (
    QualifiedBreach,
    TransitionReceipt,
    TransitionRefused,
)
from facdrone_kernel.lifecycle_monitor import (
    BaselineView,
    MonitorEvidence,
    MonitorReading,
    assess_lifecycle,
)

FlagWriter = Callable[[str, int, QualifiedBreach, int, str], TransitionReceipt]
Emit = Callable[..., None]  # (event, message, **fields): the standard's emitter
NOT_RUN = "window"  # the monitor name a not-run window is recorded under
Readings = Mapping[str, tuple[MonitorReading, ...]]  # window id -> its readings


@dataclass(frozen=True)
class MonitoredSignal:
    """One admitted signal version as the root describes it: its baseline as projected,
    the digest that baseline is cited by, the standing revision a flag is expected at,
    and the moment its windows are laid from (the admission)."""

    signal: str
    version: int
    baseline: BaselineView
    baseline_digest: str
    expected_revision: int
    anchor: datetime


Evidence = Callable[[MonitoredSignal, Window], MonitorEvidence | str]


@dataclass(frozen=True)
class MonitorSummary:
    """What one run did (R10.4): windows evaluated, flag event ids appended, the counts
    of watch, unmeasured and not-run outcomes, and ``unavailable`` naming every signal
    whose cycle failed with why: lost coverage, never health."""

    windows_evaluated: int
    flags: tuple[str, ...]
    watch: int
    unmeasured: int
    not_run: int
    unavailable: tuple[tuple[str, int, str], ...] = ()


@dataclass(frozen=True)
class MonitorSources:
    """What a composition root binds the runner to: what is monitored, where a window's
    evidence comes from, and the flag writer bound into a transaction for a decision
    (R4.3, R12.2)."""

    monitored: Callable[[], Sequence[MonitoredSignal]]
    evidence: Callable[[Sequence[Window]], Evidence]
    flag_in: Callable[[Connection, str], FlagWriter]


@dataclass(frozen=True)
class _Outcome:
    """One signal's cycle: fresh readings per window, not-run windows with their reasons,
    watches per monitor, and the flag if one qualified."""

    windows: Readings
    not_run: tuple[tuple[str, str], ...]
    watch: tuple[tuple[str, int], ...]
    flag: TransitionReceipt | None
    monitor: str | None
    reason: str | None


@dataclass(frozen=True)
class _Cycle:
    """One signal inside one window's transaction, at the run's injected moment."""

    target: MonitoredSignal
    decision_id: str
    as_of: datetime
    conn: Connection


@dataclass(frozen=True)
class _Plan:
    target: MonitoredSignal
    windows: tuple[Window, ...]
    recorded: list[tuple[str, str, str]]


def _unavailable(
    say: Emit, target: MonitoredSignal, failure: Exception, decision: str
) -> tuple[str, int, str]:
    why = f"{type(failure).__name__}: {failure}"
    say(
        "lifecycle.monitor_unavailable",
        "monitor coverage unavailable",
        signal=target.signal,
        version=target.version,
        reason=why,
        decision_id=decision,
    )
    return target.signal, target.version, why


def _reading_id(cycle: _Cycle, monitor: str, window: str, rules: str) -> str:
    t = cycle.target
    key = f"{t.signal}|{t.version}|{t.baseline_digest}|{monitor}|{window}|{rules}"
    return hashlib.sha256(key.encode()).hexdigest()


def _rules_revision(declaration: MonitorDeclaration) -> str:
    text = json.dumps({"method": "dated-ic-v2", **asdict(declaration)}, sort_keys=True)
    return hashlib.sha256(text.encode()).hexdigest()[:16]


def _flag_reason(monitor: str, count: int, windows: Sequence[str], latest: MonitorReading) -> str:
    """The reading, the threshold, the count and the windows (R5.4)."""
    threshold = f"{latest.value:.4g} {latest.comparator} {latest.boundary:.4g} {latest.units}"
    named = f"the latest {len(windows)}, {windows[0]} to {windows[-1]}, qualify"
    return f"{monitor} breached on {count} consecutive windows ({named}); last reading {threshold}"


class MonitorRunner:
    """Bound once by a composition root; called once per served decision."""

    def __init__(self, engine: Engine, declaration: MonitorDeclaration, sources: MonitorSources):
        self._engine = engine
        self._declaration = declaration
        self._monitored = sources.monitored
        self._evidence = sources.evidence
        self._flag_in = sources.flag_in
        self._rules = _rules_revision(declaration)

    def run(self, as_of: datetime, decision_id: str, emit: Emit | None = None) -> MonitorSummary:
        """Every due window commits independently; the final window includes the flag
        over the full recorded + fresh sequence, then emission (R12.6); a failed signal is
        named unavailable and the remaining signals still run."""
        say: Emit = emit or (lambda *_a, **_k: None)
        evaluated, watch, unmeasured, not_run = 0, 0, 0, 0
        flags: list[str] = []
        lost: list[tuple[str, int, str]] = []
        plans = []
        for target in self._monitored():
            try:
                plans.append(self._plan(target, as_of))
            except Exception as failure:  # noqa: BLE001 — persistence down: coverage lost
                lost.append(_unavailable(say, target, failure, decision_id))
        evidence = self._evidence(tuple(w for p in plans for w in p.windows))
        for plan in plans:
            target = plan.target
            try:
                outcome = self._cycle(plan, as_of, decision_id, evidence)
            except Exception as failure:  # noqa: BLE001 — persistence down: coverage lost
                lost.append(_unavailable(say, target, failure, decision_id))
                continue
            self._emit(say, target, outcome, decision_id)
            evaluated += len(outcome.windows)
            unmeasured += sum(
                r.disposition == "unmeasured" for rs in outcome.windows.values() for r in rs
            )
            not_run += len(outcome.not_run)
            watch += len(outcome.watch)
            if outcome.flag is not None:
                flags.append(outcome.flag.event_id)
        return MonitorSummary(evaluated, tuple(flags), watch, unmeasured, not_run, tuple(lost))

    def _emit(
        self, say: Emit, target: MonitoredSignal, outcome: _Outcome, decision_id: str
    ) -> None:
        who = {"signal": target.signal, "version": target.version, "decision_id": decision_id}
        for window_id, readings in outcome.windows.items():
            for r in readings:
                say(
                    "lifecycle.reading",
                    "one monitor reading",
                    window=window_id,
                    monitor=r.monitor,
                    value=r.value,
                    boundary=r.boundary,
                    comparator=r.comparator,
                    units=r.units,
                    disposition=r.disposition,
                    limitation=r.limitation,
                    **who,
                )
        for window_id, reason in outcome.not_run:
            say("lifecycle.not_run", "a window did not run", window=window_id, reason=reason, **who)
        for monitor, count in outcome.watch:
            say(
                "lifecycle.watch",
                "a breach without persistence",
                monitor=monitor,
                count=count,
                **who,
            )
        if outcome.flag is not None:
            say(
                "lifecycle.flag",
                "a signal was flagged",
                monitor=outcome.monitor,
                reason=outcome.reason,
                event_id=outcome.flag.event_id,
                **who,
            )
        elif outcome.reason is not None:  # a refused flag: an expected outcome (R12.5)
            say(
                "lifecycle.refused",
                "a flag was refused",
                monitor=outcome.monitor,
                reason=outcome.reason,
                **who,
            )

    def _plan(self, target: MonitoredSignal, as_of: datetime) -> _Plan:
        complete = as_of.date() - timedelta(days=1)  # a window is due the day after its cutoff
        windows = scheduled_windows(self._declaration.runner, target.anchor.date(), complete)
        with self._engine.connect() as conn:
            recorded = self._recorded(_Cycle(target, "", as_of, conn))
        return _Plan(target, due_windows(windows, sorted({w for w, _, _ in recorded})), recorded)

    def _cycle(
        self, plan: _Plan, as_of: datetime, decision_id: str, evidence_for: Evidence
    ) -> _Outcome:
        fresh: dict[str, tuple[MonitorReading, ...]] = {}
        not_run: list[tuple[str, str]] = []
        for window in plan.windows:
            evidence = evidence_for(plan.target, window)
            readings = (
                ()
                if isinstance(evidence, str)
                else assess_lifecycle(evidence, plan.target.baseline, self._declaration.rules)
            )
            with self._engine.begin() as conn:
                cycle = _Cycle(plan.target, decision_id, as_of, conn)
                self._append(
                    cycle,
                    window.window_id,
                    readings,
                    evidence if isinstance(evidence, str) else None,
                )
                if isinstance(evidence, str):
                    not_run.append((window.window_id, evidence))
                else:
                    fresh[window.window_id] = readings
                if window == plan.windows[-1]:
                    return self._finish(cycle, plan.recorded, fresh, not_run)
        with self._engine.begin() as conn:
            return self._finish(
                _Cycle(plan.target, decision_id, as_of, conn), plan.recorded, fresh, not_run
            )

    def _finish(
        self,
        cycle: _Cycle,
        recorded: list[tuple[str, str, str]],
        fresh: Readings,
        not_run: Sequence[tuple[str, str]],
    ) -> _Outcome:
        # Judge the same final sequence as before; commit its flag with the last window.
        # A crash before the flag commits therefore leaves that last window due again.
        sequence = recorded + [(w, r.monitor, r.disposition) for w, rs in fresh.items() for r in rs]
        sequence += [(w, NOT_RUN, "not_run") for w, _ in not_run]
        flag, monitor, reason, watch = self._judge(cycle, sequence, fresh)
        return _Outcome(fresh, tuple(not_run), tuple(watch), flag, monitor, reason)

    def _recorded(self, cycle: _Cycle) -> list[tuple[str, str, str]]:
        """The readings already on record for this signal version under this baseline and
        these rules: the sequence a new window joins."""
        t = SIGNAL_MONITOR_READING
        rows = cycle.conn.execute(
            select(t.c.window_id, t.c.monitor, t.c.disposition)
            .where(
                t.c.signal == cycle.target.signal,
                t.c.version == cycle.target.version,
                t.c.baseline_digest == cycle.target.baseline_digest,
                t.c.rules_revision == self._rules,
            )
            .order_by(t.c.window_id)
        ).all()
        return [(w, m, d) for w, m, d in rows]

    def _append(
        self, cycle: _Cycle, window: str, readings: Sequence[MonitorReading], not_run: str | None
    ) -> None:
        """Insert the window's rows; a row already present at this id is left as it is."""
        rows = [self._row(cycle, window, r, None) for r in readings]
        if not_run is not None:
            absent = MonitorReading(NOT_RUN, None, None, "", "", 0, 0, "unmeasured", not_run)
            rows.append(self._row(cycle, window, absent, "not_run"))
        if rows:
            statement = insert(SIGNAL_MONITOR_READING).values(rows)
            cycle.conn.execute(statement.on_conflict_do_nothing(index_elements=["reading_id"]))

    def _row(
        self, cycle: _Cycle, window: str, r: MonitorReading, disposition: str | None
    ) -> dict[str, object]:
        return {
            "reading_id": _reading_id(cycle, r.monitor, window, self._rules),
            "signal": cycle.target.signal,
            "version": cycle.target.version,
            "monitor": r.monitor,
            "baseline_digest": cycle.target.baseline_digest,
            "rules_revision": self._rules,
            "window_id": window,
            "cutoff": datetime.fromisoformat(window).replace(tzinfo=UTC),
            "value": r.value,
            "boundary": r.boundary,
            "comparator": r.comparator,
            "units": r.units,
            "valid": r.disposition in ("breach", "within"),
            "disposition": disposition or r.disposition,
            "decision_id": cycle.decision_id,
            "known_at": cycle.as_of,  # the injected moment the window was judged at
            "payload": {
                "observations": r.observations,
                "effective_observations": r.effective_observations,
                "limitation": r.limitation,
            },
        }

    def _judge(
        self, cycle: _Cycle, sequence: Sequence[tuple[str, str, str]], fresh: Readings
    ) -> tuple[TransitionReceipt | None, str | None, str | None, list[tuple[str, int]]]:
        """Persistence per monitor over distinct windows (R5.4, R5.5): a window the
        monitor has no reading for breaks its run; every monitor's watch is kept; the
        first monitor whose latest qualifying window this run appended flags once, inside
        the same transaction as the readings (R12.6), so a repeated serve raises nothing
        again while a run that continues after a refusal is put to the writer at every
        new window; a refused flag (the standing moved) is on record."""
        count = self._declaration.runner.persistence_count
        windows = sorted({w for w, _, _ in sequence})
        watch: list[tuple[str, int]] = []
        candidate: tuple[str, str, str, tuple[str, ...]] | None = None
        for monitor in sorted({m for _, m, _ in sequence if m != NOT_RUN}):
            by_window = {w: d for w, m, d in sequence if m == monitor}
            verdict = persistence([(w, by_window.get(w, "not_run")) for w in windows], count)
            if verdict.disposition == "watch":
                watch.append((monitor, verdict.count))
            if verdict.disposition == "flag" and verdict.windows[-1] in fresh and not candidate:
                latest = next(r for r in fresh[verdict.windows[-1]] if r.monitor == monitor)
                reason = _flag_reason(monitor, verdict.count, verdict.windows, latest)
                candidate = (monitor, reason, verdict.windows[-1], verdict.windows)
        if candidate is None:
            return None, None, None, watch
        monitor, reason, _, named = candidate
        ids = tuple(_reading_id(cycle, monitor, w, self._rules) for w in named)
        target = cycle.target
        try:
            flag = self._flag_in(cycle.conn, cycle.decision_id)
            receipt = flag(
                target.signal,
                target.version,
                QualifiedBreach(monitor, ids, named, reason),
                target.expected_revision,
                reason,
            )
        except TransitionRefused as refused:
            return None, monitor, f"refused: {refused}", watch
        return receipt, monitor, reason, watch
```

### src/facdrone/shells/lifecycle_store.py

```python
"""The one lifecycle transition authority (facdrone-signal-lifecycle 9.1; R4.2 to R4.4,
R4.6, R4.7, R9.1, R9.3, R12.6): append-only events in PostgreSQL, one private
persistence behind three purpose-bound writers (admit, flag, retire) and the enrollment
act. Each writer has a fixed target state and validates the prior state, the authority
and the evidence through the domain's transition rule before it writes; a refused attempt
is written as a refused event at the revision it met, then raised. Expected-revision
conflicts serialize competing writers: applied events are unique per (signal, version,
revision), so two writers at one revision cannot both apply. Standing and open flags are
derived from the applied events; there is no second flag table. Only the composition
roots may import this module (import-linter); a caller receives one bound writer, never
the store. Nothing here emits: the event is committed before any emitter could run.
"""

from __future__ import annotations

import hashlib
import json
from collections.abc import Callable
from dataclasses import asdict, dataclass, replace
from datetime import UTC, datetime, timedelta

from sqlalchemy import Connection, Engine, insert, select
from sqlalchemy.exc import IntegrityError

from facdrone.schema import SIGNAL_LIFECYCLE_EVENT, SIGNAL_MONITOR_READING
from facdrone_kernel.domain.signal_lifecycle import (
    AdmissionEvidence,
    Enrollment,
    Evidence,
    LifecycleEvent,
    LifecycleState,
    OpenFlag,
    QualifiedBreach,
    RecordedHumanRetirement,
    Standing,
    StandingRevision,
    TransitionReceipt,
    TransitionRefused,
    UnrecordedStanding,
    refusal_reason,
)


@dataclass(frozen=True)
class TransitionRequest:
    """One attempt: the fixed target and authority a writer supplies, the evidence and the
    expected standing revision the caller supplies."""

    signal: str
    version: int
    target: LifecycleState
    authority: str
    evidence: Evidence
    expected_revision: int
    reason: str
    decision_id: str | None = None  # the decision whose cycle raised it (R12.2)


def _event_id(event: LifecycleEvent) -> str:
    """The content digest, with the moment in UTC so no session offset enters it."""
    content = {**asdict(event), "event_at": event.event_at.astimezone(UTC).isoformat()}
    content.pop("event_id")
    return hashlib.sha256(json.dumps(content, sort_keys=True).encode()).hexdigest()


class LifecycleStore:
    """Private persistence; nothing outside a composition root sees it."""

    def __init__(self, engine: Engine, now: Callable[[], datetime] = lambda: datetime.now(UTC)):
        self._engine = engine
        self._now = now

    def events(self, signal: str, version: int) -> tuple[LifecycleEvent, ...]:
        """Every event of one signal version, applied and refused, in knowledge order, each
        carrying the event id the store wrote."""
        t = SIGNAL_LIFECYCLE_EVENT
        query = (
            select(t)
            .where(t.c.signal == signal, t.c.version == version)
            .order_by(t.c.known_at, t.c.revision)
        )
        with self._engine.connect() as conn:
            rows = conn.execute(query).mappings().all()
        names = [
            c.name for c in t.columns if c.name not in ("known_at", "prior_state", "new_state")
        ]
        return tuple(
            LifecycleEvent(
                **{n: r[n] for n in names},
                prior_state=LifecycleState(r["prior_state"]) if r["prior_state"] else None,
                new_state=LifecycleState(r["new_state"]),
            )
            for r in rows
        )

    def _applied(self, signal: str, version: int) -> tuple[LifecycleEvent, ...]:
        return tuple(e for e in self.events(signal, version) if e.disposition == "applied")

    def standing(self, signal: str, version: int) -> Standing:
        """The latest applied event as the standing; unrecorded when none (R4.8)."""
        applied = self._applied(signal, version)
        if not applied:
            return UnrecordedStanding(signal, version)
        last = max(applied, key=lambda e: e.revision)
        census = last.evidence_ref if last.authority == "census" else None
        return StandingRevision(
            signal, version, last.new_state, last.revision, last.event_at, census
        )

    def census_references(self, signal: str, version: int) -> tuple[str, ...]:
        """Every verdict an applied census event of this version has admitted on (R4.7): a
        verdict already cited admits nothing again, whatever the standing is now."""
        applied = self._applied(signal, version)
        return tuple(e.evidence_ref for e in applied if e.authority == "census")

    def admitted(self) -> tuple[StandingRevision, ...]:
        """Every signal version whose standing is admitted (lifecycle 10.2): what the
        monitor watches; a flagged or unrecorded version has no windows until a census."""
        t = SIGNAL_LIFECYCLE_EVENT
        with self._engine.connect() as conn:
            pairs = conn.execute(select(t.c.signal, t.c.version).distinct()).all()
        standings = (self.standing(s, v) for s, v in pairs)
        return tuple(
            s
            for s in standings
            if isinstance(s, StandingRevision) and s.state is LifecycleState.ADMITTED
        )

    def open_flags(self) -> tuple[OpenFlag, ...]:
        """Every signal version whose latest applied event is a flag: the flag store is
        that subset, and the flag id is the id the store wrote (R9.1, R9.3)."""
        t = SIGNAL_LIFECYCLE_EVENT
        with self._engine.connect() as conn:
            pairs = conn.execute(select(t.c.signal, t.c.version).distinct()).all()
        flags = []
        for signal, version in pairs:
            applied = self._applied(signal, version)
            if applied and (last := max(applied, key=lambda e: e.revision)).new_state == "flagged":
                flags.append(
                    OpenFlag(
                        signal, version, last.event_id, last.reason, last.event_at, last.decision_id
                    )
                )
        return tuple(flags)

    def overdue_flags(self, after: timedelta) -> tuple[OpenFlag, ...]:
        now = self._now()
        return tuple(f for f in self.open_flags() if f.overdue(now, after))

    def readings(self, signal: str, version: int) -> tuple[dict[str, object], ...]:
        """Every reading recorded for the version, in window then monitor order (11.3,
        R10.3): what the analyst surface publishes, never recomputed."""
        t = SIGNAL_MONITOR_READING
        with self._engine.connect() as conn:
            rows = conn.execute(
                select(t)
                .where(t.c.signal == signal, t.c.version == version)
                .order_by(t.c.window_id, t.c.monitor)
            ).mappings()
            return tuple(dict(r) for r in rows)

    def _write(self, event: LifecycleEvent, connection: Connection | None = None) -> str:
        """Append one event: in the caller's transaction when it supplies one (the monitor
        appends its readings and its flag together, R12.6), else in its own."""
        stamped = replace(event, event_id=_event_id(event))
        row = {**asdict(stamped), "known_at": self._now()}
        row["prior_state"] = event.prior_state.value if event.prior_state else None
        row["new_state"] = event.new_state.value
        if connection is not None:
            connection.execute(insert(SIGNAL_LIFECYCLE_EVENT).values(**row))
            return stamped.event_id
        with self._engine.begin() as conn:
            conn.execute(insert(SIGNAL_LIFECYCLE_EVENT).values(**row))
        return stamped.event_id

    def _refuse(self, event: LifecycleEvent, met: int, why: str) -> TransitionRefused:
        """Record the refused attempt at the revision it met (closing nothing), then hand
        back the refusal; an identical attempt at an identical injected moment is already
        on record, so its repeat is not a second row."""
        refused = replace(
            event,
            revision=met,
            reason=f"refused: {why}",
            disposition="refused",
            closed_flag_id=None,
        )
        try:
            self._write(refused)
        except IntegrityError:  # the same content digest: this refusal is already recorded
            pass
        return TransitionRefused(f"{event.signal} v{event.version} -> {event.new_state}: {why}")

    def transition(
        self, request: TransitionRequest, connection: Connection | None = None
    ) -> TransitionReceipt:
        """Apply one transition at the expected revision or record why it was refused; a
        refusal is always recorded in its own transaction so it survives a rolled-back
        caller. A concurrent write at the same revision inside a caller's transaction has
        already aborted that transaction, so it is re-raised for the caller to fail on
        (the monitor then reports the signal unavailable) rather than recorded beside
        readings that will not commit."""
        r = request
        standing = self.standing(r.signal, r.version)
        current = standing.revision if isinstance(standing, StandingRevision) else 0
        prior = standing.state if isinstance(standing, StandingRevision) else None
        closes = self._flag_id(standing) if prior == LifecycleState.FLAGGED else None
        event = LifecycleEvent(
            r.signal,
            r.version,
            prior,
            r.target,
            r.expected_revision + 1,
            self._now(),
            r.authority,
            r.reason,
            r.evidence.reference(),
            "applied",
            closes,
            decision_id=r.decision_id,
        )
        if current != r.expected_revision:
            why = f"expected revision {r.expected_revision}, standing is at {current}"
            raise self._refuse(event, current, why)
        if (blocked := refusal_reason(standing, r.target, r.authority, r.evidence)) is not None:
            raise self._refuse(event, current, blocked)
        try:
            event_id = self._write(event, connection)
        except IntegrityError as clash:  # only the partial unique index can raise here
            if connection is not None:  # the caller's transaction is aborted: fail loudly
                raise
            why = f"revision {event.revision} was written concurrently ({type(clash).__name__})"
            raise self._refuse(event, current, why) from None
        return TransitionReceipt(
            event_id, r.signal, r.version, prior, r.target, event.revision, closes
        )

    def _flag_id(self, standing: Standing) -> str | None:
        if not isinstance(standing, StandingRevision):
            return None
        applied = self._applied(standing.signal, standing.version)
        at = [e for e in applied if e.revision == standing.revision]
        return at[0].event_id if at else None


@dataclass(frozen=True)
class LifecycleWriters:
    """The purpose-bound callables a composition root hands out one at a time (R4.3): each
    has a fixed target and authority; there is no set_state."""

    enroll: Callable[[str, int, Enrollment, str], TransitionReceipt]
    admit: Callable[[str, int, AdmissionEvidence, int, str], TransitionReceipt]
    flag: Callable[[str, int, QualifiedBreach, int, str], TransitionReceipt]
    retire: Callable[[str, int, RecordedHumanRetirement, int, str], TransitionReceipt]


def _bound(
    store: LifecycleStore,
    target: LifecycleState,
    authority: str,
    cycle: tuple[Connection | None, str | None],
) -> Callable[[str, int, Evidence, int, str], TransitionReceipt]:
    connection, decision_id = cycle

    def act(
        signal: str, version: int, evidence: Evidence, expected: int, reason: str
    ) -> TransitionReceipt:
        request = TransitionRequest(
            signal, version, target, authority, evidence, expected, reason, decision_id
        )
        return store.transition(request, connection)

    return act


def bind_writers(
    store: LifecycleStore, connection: Connection | None = None, decision_id: str | None = None
) -> LifecycleWriters:
    """Bind the four acts to one store; a consumer receives only the one it is permitted.
    With ``connection`` the acts append inside that transaction (the monitor's readings
    and flag land together); a refusal is still recorded on its own. ``decision_id`` is
    stamped on every event the acts write: the decision whose cycle raised it (R12.2)."""
    cycle = (connection, decision_id)
    enroll = _bound(store, LifecycleState.CANDIDATE, "registry", cycle)
    return LifecycleWriters(
        lambda signal, version, act, reason: enroll(signal, version, act, 0, reason),
        _bound(store, LifecycleState.ADMITTED, "census", cycle),
        _bound(store, LifecycleState.FLAGGED, "monitor", cycle),
        _bound(store, LifecycleState.RETIRED, "human", cycle),
    )
```

```python
    and evidence cutoff the reading was judged on, compared with the recorded rows; a

    from facdrone.shells.cli.sim_group import MANDATE
    from facdrone.shells.cppi.snapshot_sources import readings_source

    version = int(version_text)
    store = _lifecycle_store()
    stored = [r for r in store.readings(name, version) if r["window_id"] == window]

        typer.echo(f"no reading recorded for {target} in window {window}", err=True)
        raise typer.Exit(2)
    declared = monitor_declaration(
        read_parameters(project_root() / "conf" / "base" / "parameters.yml")
    )
    stored = [r for r in stored if str(r["rules_revision"]) == _rules_revision(declared)]
    baseline = _cited_baseline(store, name, version)
    same_basis = baseline is not None and all(
        str(baseline["digest"]) == str(r["baseline_digest"]) for r in stored
    )
    if not stored or not same_basis:
        typer.echo("not reproducible as published: the baseline or the rules changed", err=True)
        raise typer.Exit(2)
    assert baseline is not None  # noqa: S101 — established by same_basis
    frame = _window(window, declared.runner.trailing_days)
    registry = configured_pods(record=MANDATE)
    pod = registry.get_scored(_pod_of(name, registry))
    identity = (name, version, int(baseline["traded_horizon_days"]))
    ic_days = declared.runner.ic_trailing_days
    readings = readings_source(store._engine, pod.dynamics)  # noqa: SLF001
    evidence = live_evidence(store._engine, readings, identity, frame, ic_days)  # noqa: SLF001
    if isinstance(evidence, str):
        typer.echo(f"not run: {evidence}")
        raise typer.Exit(0 if any(r["disposition"] == "not_run" for r in stored) else 1)
    if all(r["monitor"] == "window" for r in stored):
        typer.echo("DIFFERS: recorded as not run, but the window is measurable now")
        raise typer.Exit(1)
    fresh = assess_lifecycle(evidence, baseline_view(baseline), declared.rules)

def _cited_baseline(store: LifecycleStore, name: str, version: int) -> dict[str, Any] | None:
    """The ledger baseline the version's latest applied admission cites, if any."""

    return dict(cited["baseline"]) if cited else None


def _window(window_id: str, trailing_days: int) -> Window:
    from datetime import timedelta

    from facdrone.pipelines.evaluation.monitor_windows import Window

    cutoff = date.fromisoformat(window_id)
    return Window(window_id, cutoff - timedelta(days=trailing_days), cutoff)


def _compare(stored: Sequence[Mapping[str, Any]], fresh: Mapping[str, Any]) -> int:
    """Print each recorded monitor against its re-measurement; 1 when any differs."""
    differs = 0
    for row in (r for r in stored if r["monitor"] != "window"):  # the not-run marker row
        again = fresh.get(str(row["monitor"]))
        now = (getattr(again, "value", None), getattr(again, "boundary", None))
        same = again is not None and now == (row["value"], row["boundary"])
        same = same and getattr(again, "disposition", None) == row["disposition"]
        differs += not same
        verdict = "reproduced" if same else "DIFFERS"
        typer.echo(f"{row['monitor']:24} {verdict}  was {row['disposition']} {row['value']}")
    return 1 if differs else 0


def _pod_of(signal: str, registry: RegisteredPods) -> str:
    """The scored pod that declares the signal in one of its sleeves."""
    for pod_name in registry.scored_names():
        pod = registry.get_scored(pod_name)
        buckets = (b for s in pod.sleeves for b in (s.trending, s.mean_reverting, s.neutral))
        if any(signal in bucket for bucket in buckets):
            return pod_name
    raise typer.BadParameter(f"{signal} is declared in no scored pod")
```

### src/facdrone/shells/lifecycle_evidence.py

```python
"""What the live path presents to the lifecycle monitor (facdrone-signal-lifecycle 10.2;
R5.1, R5.9, R6.1, R6.2, R6.5, R7.4, R7.7, R12.1). Two projections, both from retained
evidence and both honest about what the record does not hold: the admission baseline as
the verdict ledger keeps it, and one window's evidence from the IC the combine consumed
(Requirement 1's retained path) joined to the market-state readings the decision loop
read. A cost component the census never priced is absent, so the cost reading stays
unmeasured; the under-mandate contribution is not attributed per signal on the live path,
and the survival reading says so. IC keeps first consumption; sign/gap use the consumed window.
"""

from __future__ import annotations

from collections import Counter
from collections.abc import Callable, Mapping, Sequence
from dataclasses import dataclass, replace
from datetime import UTC, date, datetime, timedelta
from statistics import fmean
from typing import Any, Protocol

from sqlalchemy import Engine, Select, select

from facdrone.pipelines.evaluation.monitor_windows import Window
from facdrone.schema import DECISION_EVIDENCE
from facdrone.shells.decision_evidence_store import decode_consumed_evidence
from facdrone_kernel.domain import decision_evidence as values
from facdrone_kernel.domain.enums import MarketState
from facdrone_kernel.lifecycle_monitor import BaselineView, MonitorEvidence

Row = dict[str, Any]


class _EntryReading(Protocol):
    @property
    def state(self) -> MarketState: ...


class _DayReadings(Protocol):
    @property
    def entry(self) -> _EntryReading: ...


# The market-state readings through a day, bound by the caller over its engine and dynamics and
# typed by the one value the monitor reads (sim.market_state_series.DayReadings satisfies it).
Readings = Callable[[date], Mapping[date, _DayReadings]]
CONTRIBUTION_LIMITATION = "under-mandate contribution is not attributed per signal live"


def baseline_view(baseline: Mapping[str, Any]) -> BaselineView:
    """The ledger's baseline row as the monitor compares against it: only what the census
    recorded; a component it did not price is None and reads unmeasured."""
    traded = baseline.get("traded_ic")
    states = baseline.get("states") or ()
    cost = baseline.get("cost") or {}
    premise = baseline.get("premise") or {}
    return BaselineView(
        route=str(baseline["route"]),
        traded_horizon_days=int(baseline["traded_horizon_days"]),
        traded_ic_mean=None if not isinstance(traded, Mapping) else float(traded["mean"]),
        survival_ratio=(baseline.get("survival") or {}).get("ratio"),
        state_shares={str(s["state"]): float(s["share"]) for s in states},
        state_ic={str(s["state"]): s.get("ic_mean") for s in states},
        cost_basis={
            "spread": None,
            "slippage": None,
            "financing": cost.get("financing_annual_rate"),
            "dividend": None,
        },
        cost_units="annual rate, share of notional",
        premise_untested=tuple(
            str(name) for name, status in premise.get("sensitivities", ()) if status != "tested"
        ),
    )


def _consumed_query(since: datetime, through: datetime) -> Select[Any]:
    t = DECISION_EVIDENCE
    return (
        select(t.c.payload["identity"].label("identity"), t.c.payload["consumed"].label("consumed"))
        .where(t.c.scope == "live", t.c.knowledge_at > since, t.c.knowledge_at <= through)
        .order_by(t.c.knowledge_at, t.c.step_id)
        .execution_options(stream_results=True, yield_per=32)
    )


def _consumed(engine: Engine, since: datetime, through: datetime) -> list[values.ConsumedIC]:
    """Stream only identity and ICs; the large input slices never leave PostgreSQL."""
    consumed: list[values.ConsumedIC] = []
    with engine.connect() as conn:  # closed before returning: no transaction outlives the read
        for row in conn.execute(_consumed_query(since, through)).mappings():
            batch = decode_consumed_evidence(dict(row))
            consumed.extend(replace(c, knowledge_at=batch.identity.as_of) for c in batch.consumed)
    return consumed


def _observed_day(day: date) -> date:
    if isinstance(day, datetime):
        if day.utcoffset() != timedelta(0) or day.time() != datetime.min.time():
            raise ValueError("ambiguous observed session date")
        return day.date()
    return day


def _daily(consumed: list[values.ConsumedIC], window: Window) -> dict[date, tuple[float, datetime]]:
    series: dict[date, tuple[float, datetime]] = {}
    for c in consumed:
        known = c.knowledge_at
        if not isinstance(known, datetime) or known.tzinfo is None:
            raise ValueError("missing dated consumption endpoint")
        known = known.astimezone(UTC)
        for v in c.window if isinstance(c.window, tuple) else ():
            day = _observed_day(v.observed_at)
            if day > known.date():
                raise ValueError("observed session date follows consumption")
            if not window.start <= day <= window.cutoff:
                continue
            point = (float(v.value), known)
            old = series.get(day, point)
            if old[1] == known and old != point:
                raise ValueError("conflicting IC at first consumption")
            series[day] = min(old, point, key=lambda p: p[1])
    return dict(sorted(series.items()))


def _lookback(
    traded: list[values.ConsumedIC], start: date
) -> tuple[values.DatedICValue, ...] | values.AuditedAbsence | None:
    recent = {
        c.knowledge_at: c.window
        for c in traded
        if isinstance(c.knowledge_at, datetime) and c.knowledge_at.astimezone(UTC).date() >= start
    }
    latest = max(recent, default=None)
    lookback = recent.get(latest) if latest is not None else None
    if any(c.knowledge_at == latest and c.window != lookback for c in traded):
        raise ValueError("conflicting latest consumed lookback")
    return lookback


def read_evidence(
    engine: Engine, readings: Readings, windows: Sequence[Window], ic_days: int
) -> LiveEvidence:
    """Retain one compact IC sequence for this run's union of due windows."""
    consumed: list[values.ConsumedIC] = []
    if windows:
        since = datetime.combine(min(w.cutoff for w in windows), datetime.min.time(), UTC)
        through = datetime.combine(max(w.cutoff for w in windows), datetime.max.time(), UTC)
        consumed = _consumed(engine, since - timedelta(days=ic_days + 1), through)
    return LiveEvidence(readings, consumed, ic_days)


def live_evidence(
    engine: Engine, readings: Readings, signal: tuple[str, int, int], window: Window, ic_days: int
) -> MonitorEvidence | str:
    """The standalone reproduction path uses the same projection as a monitor run."""
    return read_evidence(engine, readings, (window,), ic_days).for_window(signal, window)


@dataclass(frozen=True)
class LiveEvidence:
    """One run's retained ICs; no cache survives the run or a changed archive."""

    readings: Readings
    consumed: list[values.ConsumedIC]
    ic_days: int

    def for_window(self, signal: tuple[str, int, int], window: Window) -> MonitorEvidence | str:
        """Apply the original signal, horizon and knowledge bounds to retained ICs."""
        name, version, horizon = signal
        ic_window = replace(window, start=window.cutoff - timedelta(days=self.ic_days))
        since = datetime.combine(ic_window.start, datetime.min.time(), UTC)
        through = datetime.combine(window.cutoff, datetime.max.time(), UTC)
        mine = [
            c
            for c in self.consumed
            if (c.signal.name, c.signal.version) == (name, version)
            and isinstance(c.knowledge_at, datetime)
            and c.knowledge_at.tzinfo is not None
            and since <= c.knowledge_at <= through
        ]
        traded = [c for c in mine if c.horizon_days == horizon]
        series = _daily(traded, ic_window)
        lookback = _lookback(traded, window.start)
        if not series and lookback is None:
            return f"no IC consumed at the traded horizon ({horizon}d) in window {window.window_id}"
        readings: Mapping[Any, _DayReadings] = self.readings(window.cutoff)
        state_of = {
            d: r.entry.state.value
            for d, r in readings.items()
            if ic_window.start <= d <= window.cutoff
        }
        by_state: dict[str, list[float]] = {}
        state_intervals: dict[str, list[tuple[date, date]]] = {}
        for day, (value, known) in series.items():
            state = state_of.get(day)
            if state is not None:
                by_state.setdefault(state, []).append(value)
                state_intervals.setdefault(state, []).append((day, known.date()))
        horizons = sorted({c.horizon_days for c in mine})
        by_horizon = {
            h: [v for v, _ in _daily([c for c in mine if c.horizon_days == h], ic_window).values()]
            for h in horizons
        }
        return MonitorEvidence(
            ic_values=tuple(v for v, _ in series.values()),
            lookback_ic=tuple(v.value for v in lookback) if isinstance(lookback, tuple) else (),
            dwell_days=dict(Counter(s for d, s in state_of.items() if d >= window.start)),
            state_ic={s: tuple(v) for s, v in by_state.items()},
            cost={"spread": None, "slippage": None, "financing": None, "dividend": None},
            contribution=None,
            standalone=None,
            contribution_contaminated=CONTRIBUTION_LIMITATION,
            ic_by_horizon={h: fmean(v) for h, v in by_horizon.items() if v},
            ic_intervals=tuple((d, known.date()) for d, (_, known) in series.items()),
            state_intervals={s: tuple(v) for s, v in state_intervals.items()},
        )
```

### src/facdrone/shells/lifecycle_projection.py

```python
"""The published lifecycle standing an analyst reads (facdrone-signal-lifecycle 11.3; R10.1
to R10.5, R9.2, R9.4). Pure over what the store and the ledger already hold: the recorded
standing, the admission baseline the standing cites, the readings the monitor recorded
(the latest per monitor, and every window it did not run), and the open flag with its
age. Every figure here was measured and judged before it is projected; nothing is computed
at render time, and every name a reader meets is an analyst label, never a code
identifier. An absence is stated as an absence: no baseline, no reading, a monitor that
did not run and why; a signal is never shown healthy on the strength of silence.
"""

from __future__ import annotations

from collections.abc import Mapping, Sequence
from datetime import datetime, timedelta
from typing import Any

from facdrone_kernel.domain.signal_lifecycle import OpenFlag, Standing, StandingRevision

Row = Mapping[str, Any]

STANDING_LABELS = {
    "candidate": "Candidate: not in the book",
    "admitted": "Admitted",
    "flagged": "Flagged: earning the no-edge weight",
    "retired": "Retired",
}
DISPOSITION_LABELS = {
    "breach": "Breached",
    "within": "Within its boundary",
    "unmeasured": "Not measured",
    "descriptive": "Descriptive",
    "not_run": "Did not run",
}
MONITOR_LABELS = {
    "traded_horizon_ic": "IC at the traded horizon",
    "survival": "Survival under the mandate",
    "sign_stability": "Sign stability",
    "ic_gap": "Peak minus mean IC",
    "decay_half_life": "Decay half-life",
    "horizon_mismatch": "Horizon gap",
    "dwell_distance": "Regime dwell distance",
    "macro_premise": "Macro premise",
}


def monitor_label(monitor: str) -> str:
    """The analyst label of a monitor key as the library writes it: the per-state and
    per-component monitors are templated ``state_ic:<state>`` and ``cost:<component>``."""
    kind, sep, rest = monitor.partition(":")
    if sep and kind == "state_ic":
        return f"IC in the {rest} state"
    if sep and kind == "cost":
        return f"Cost drift: {rest}"
    return MONITOR_LABELS.get(monitor, monitor.replace("_", " "))


def labeled_reason(reason: str) -> str:
    """A flag reason as the runner wrote it, with the leading monitor key labeled."""
    head, sep, tail = reason.partition(" breached")
    return f"{monitor_label(head)}{sep}{tail}" if sep else reason


def _latest_basis(readings: Sequence[Row]) -> list[Row]:
    """Only the rows judged under the newest basis (baseline digest and rules revision)
    the version has readings for: a superseded basis never shows as the latest reading."""
    if not readings:
        return []
    newest = readings[-1]
    basis = (newest.get("baseline_digest"), newest.get("rules_revision"))
    return [r for r in readings if (r.get("baseline_digest"), r.get("rules_revision")) == basis]


def _baseline(row: Row | None) -> dict[str, Any] | None:
    if row is None:
        return None
    traded, survival = row.get("traded_ic"), row.get("survival") or {}
    return {
        "route": str(row.get("route")),
        "traded_horizon_days": row.get("traded_horizon_days"),
        "ic_mean": None if not isinstance(traded, Mapping) else traded.get("mean"),
        "survival_ratio": survival.get("ratio"),
        "states": [str(s.get("state")) for s in row.get("states") or ()],
        "digest": str(row.get("digest", ""))[:16],
    }


def _reading(r: Row) -> dict[str, Any]:
    payload = r.get("payload") or {}
    return {
        "monitor": monitor_label(str(r["monitor"])),
        "key": str(r["monitor"]),
        "window": str(r["window_id"]),
        "value": r.get("value"),
        "boundary": r.get("boundary"),
        "comparator": r.get("comparator") or "",
        "units": r.get("units") or "",
        "disposition": DISPOSITION_LABELS.get(str(r["disposition"]), str(r["disposition"])),
        "limitation": payload.get("limitation"),
        "judged_at": str(r.get("known_at", "")),
    }


def _latest_and_not_run(readings: Sequence[Row]) -> tuple[dict[str, Row], list[dict[str, Any]]]:
    """The last row per monitor in window order, and every window the monitor did not run."""
    latest: dict[str, Row] = {}
    not_run = []
    for r in readings:
        if r["disposition"] == "not_run":
            reason = (r.get("payload") or {}).get("limitation")
            not_run.append({"window": str(r["window_id"]), "reason": reason})
        else:
            latest[str(r["monitor"])] = r
    return latest, not_run


def _flag(flag: OpenFlag | None, now: datetime, after: timedelta) -> dict[str, Any] | None:
    if flag is None:
        return None
    return {
        "reason": labeled_reason(flag.reason),
        "opened_at": flag.opened_at.isoformat(),
        "age_days": (now - flag.opened_at).days,
        "overdue": flag.overdue(now, after),
        "event_id": flag.flag_event_id,
        "decision_id": flag.decision_id,
    }


def _state(standing: Standing) -> dict[str, Any]:
    if not isinstance(standing, StandingRevision):
        return {"standing": "No recorded standing", "since": None, "revision": None}
    return {
        "standing": STANDING_LABELS[standing.state.value],
        "since": standing.recorded_at.date().isoformat(),
        "revision": standing.revision,
    }


def signal_standing(
    standing: Standing,
    baseline: Row | None,
    readings: Sequence[Row],
    flag: OpenFlag | None,
    moment: tuple[datetime, timedelta],
) -> dict[str, Any]:
    """One signal version as the analyst surface presents it; ``moment`` is (now, the
    declared overdue age)."""
    now, after = moment
    latest, not_run = _latest_and_not_run(_latest_basis(readings))
    no_reading = "no reading recorded: the monitor has not judged this version"
    return {
        "signal": standing.signal,
        "version": standing.version,
        **_state(standing),
        "baseline": _baseline(baseline),
        "baseline_note": None if baseline is not None else "no admission baseline recorded",
        "readings": [_reading(r) for r in latest.values()],
        "readings_note": None if latest else no_reading,
        "not_run": not_run,
        "flag": _flag(flag, now, after),
    }

```

```python
"""The daemon's lifecycle seams, bound together over the root's one lifecycle store: the
post-commit monitor, which reads lifecycle evidence and writes flags in the decision's
transaction, and the time-aware status the root spreads into its binding."""

from __future__ import annotations

from collections.abc import Callable, Sequence
from datetime import datetime, timedelta
from typing import TYPE_CHECKING, TypedDict

from facdrone_kernel.lifecycle_monitor import MonitorEvidence

if TYPE_CHECKING:
    from pathlib import Path

    from facdrone.pipelines.evaluation.monitor_windows import MonitorDeclaration
    from facdrone.shells.cppi.daemon_book import _Book
    from facdrone.shells.lifecycle_monitor_runner import MonitorRunner
    from facdrone.shells.lifecycle_store import LifecycleStore


def bind_lifecycle_monitor(
    project_root: Path, book: _Book, store: LifecycleStore, declared: MonitorDeclaration
) -> MonitorRunner:
    """Bind lifecycle evidence and transactional flags; no baseline means unmonitorable."""
    from facdrone.pipelines.evaluation.census_protocol import parse_ledger
    from facdrone.pipelines.evaluation.monitor_windows import Window
    from facdrone.pipelines.evaluation.recensus import admission_evidence
    from facdrone.shells.cppi.snapshot_sources import readings_source
    from facdrone.shells.lifecycle_evidence import baseline_view, read_evidence
    from facdrone.shells.lifecycle_monitor_runner import (
        Evidence,
        MonitoredSignal,
        MonitorRunner,
        MonitorSources,
    )
    from facdrone.shells.lifecycle_store import bind_writers

    ledger = project_root / "research" / "equity_census_verdicts.jsonl"

    def monitored() -> tuple[MonitoredSignal, ...]:
        """The admitted versions whose census baseline is on record."""
        rows = parse_ledger(ledger.read_text()) if ledger.exists() else []
        baselines = {e.reference(): r for r in rows if (e := admission_evidence(r)) is not None}
        out = []
        for standing in store.admitted():
            name, version = standing.signal, standing.version
            cited = baselines.get(standing.census or "")
            if not cited:
                continue  # admitted without a baseline on record: unmonitorable, never healthy
            baseline = cited["baseline"]
            view, digest = baseline_view(baseline), str(baseline["digest"])
            out.append(
                MonitoredSignal(
                    name, version, view, digest, standing.revision, standing.recorded_at
                )
            )
        return tuple(out)

    def evidence(windows: Sequence[Window]) -> Evidence:
        readings = readings_source(book.engine_db, book.deps.policy.pod.dynamics)
        retained = read_evidence(
            book.engine_db, readings, windows, declared.runner.ic_trailing_days
        )

        def for_window(target: MonitoredSignal, window: Window) -> MonitorEvidence | str:
            identity = (target.signal, target.version, target.baseline.traded_horizon_days)
            return retained.for_window(identity, window)

        return for_window

    sources = MonitorSources(
        monitored, evidence, lambda conn, decision: bind_writers(store, conn, decision).flag
    )
    return MonitorRunner(book.engine_db, declared, sources)


class LifecycleSeams(TypedDict):
    """The daemon's two lifecycle seams, bound together over one store."""

    monitor: MonitorRunner
    lifecycle: Callable[[datetime], list[dict[str, object]]]


def bind_lifecycle(project_root: Path, book: _Book, store: LifecycleStore) -> LifecycleSeams:
    """Bind post-commit monitoring and time-aware status to the same lifecycle store."""
    from facdrone.pipelines.evaluation.monitor_windows import monitor_declaration
    from facdrone.shells.wiring.composition import read_parameters

    declared = monitor_declaration(
        read_parameters(project_root / "conf" / "base" / "parameters.yml")
    )
    after = timedelta(days=declared.runner.overdue_days)
    return LifecycleSeams(
        monitor=bind_lifecycle_monitor(project_root, book, store, declared),
        lifecycle=lambda now: [f.as_status(now, after) for f in store.open_flags()],
    )
```

```python
"""Lifecycle store and cited census baselines for operational and research commands."""

from __future__ import annotations

from datetime import datetime, timedelta
from typing import TYPE_CHECKING, Any

from facdrone.shells.wiring.composition import project_root

if TYPE_CHECKING:
    from sqlalchemy import Engine

    from facdrone.shells.lifecycle_store import LifecycleStore


def lifecycle_store(engine: Engine | None = None) -> LifecycleStore:
    """The shared store binding; a root holding the run database's engine passes it in."""
    from sqlalchemy import create_engine

    from facdrone.shells.lifecycle_store import LifecycleStore
    from facdrone.shells.wiring.composition import read_run_dsn

    credentials = project_root() / "conf" / "local" / "credentials.yml"
    return LifecycleStore(create_engine(read_run_dsn(credentials)) if engine is None else engine)


def study_baseline(store: LifecycleStore, name: str, version: int) -> dict[str, Any] | None:
    from facdrone.pipelines.evaluation.census_protocol import parse_ledger
    from facdrone.pipelines.evaluation.recensus import admission_evidence

    rows = parse_ledger((project_root() / "research" / "equity_census_verdicts.jsonl").read_text())
    baselines = {e.reference(): r for r in rows if (e := admission_evidence(r)) is not None}
    cited = [r for r in store.census_references(name, version) if r in baselines]
    return {"reference": cited[-1], "baseline": baselines[cited[-1]]["baseline"]} if cited else None


def pod_lifecycle_standing(
    store: LifecycleStore, parameters: dict[str, Any], members: set[str], now: datetime
) -> list[tuple[str, int, dict[str, Any]]]:
    """Each registered signal version in ``members`` as the analyst surface presents it."""
    from facdrone.factors.registry import production_factors
    from facdrone.pipelines.evaluation.monitor_windows import monitor_declaration
    from facdrone.shells.lifecycle_monitor_runner import _rules_revision
    from facdrone.shells.lifecycle_projection import signal_standing

    declared = monitor_declaration(parameters)
    revision = _rules_revision(declared)
    moment = (now, timedelta(days=declared.runner.overdue_days))
    flags = {(f.signal, f.version): f for f in store.open_flags()}

    def standing(name: str, version: int) -> dict[str, Any]:
        cited = study_baseline(store, name, version)
        readings = [r for r in store.readings(name, version) if r["rules_revision"] == revision]
        baseline, flag = cited["baseline"] if cited else None, flags.get((name, version))
        return signal_standing(store.standing(name, version), baseline, readings, flag, moment)

    signals = [f.metadata for f in production_factors().factors() if f.metadata.name in members]
    return [(m.name, m.version, standing(m.name, m.version)) for m in signals]
```

## Definitions

- Public API (one sentence): every name that code outside the seven regenerated files imports, reads as a module attribute, patches or overrides, with its signature and observable behaviour, including private names that tests import or patch.
- Pure core: functions that take no engine, connection or session, open none, and execute no SQL; they compute identities, digests, labels, projections, evidence series and predicates from values.
- Effect shell: functions that hold a connection or engine and perform reads or writes, build engines or stores, read files or the parameters, or drive the CLI; they call the pure core and contain no predicate logic beyond dispatch.
- Level 1 observation: the return value (dataclasses by field, including a MonitorSummary and every record the runner emits, in emission order) or the exception type, message and cause type; the ordered dumps of signal_lifecycle_event, signal_monitor_reading and decision_evidence after each operation; the files under the sequence's project root; a CLI command's exit code and its output text verbatim. Level 2 observation: SQL statement text and parameters.
- Write site: a static match of `execute|add|add_all|commit|flush|delete|merge(` or `insert|update|delete(` in the file text. Note: the `.add` rule also flags `set.add`.
- Intra-set call edge: a distinct (caller function, callee function) pair, static, by name, with both functions inside the seven scope files. Module hop: a distinct (caller file, callee file) pair of such edges with different files.

## Structural contract

Declare every function of every scope file, including methods and nested functions, by qualname in "structure" as exactly one of "pure" or "shell", as "<file>.py::<qualname>" entries, e.g. "lifecycle_store.py::LifecycleStore.standing", "lifecycle_seams.py::bind_lifecycle_monitor.monitored", "lifecycle_store.py::_bound.act".
- A declared pure function has no engine, connection or session parameter (names matching `^(engine|conn|connection|session|bind)$` or ending in `_engine`, `_conn`, `_connection`, `_session`) and makes no SQL execution call (`execute`, `begin`, `connect`, `commit`, `add`, `flush`, `merge`, `delete`); a static check verifies this for every declared pure function.
- Every write site (static match of `execute|add|add_all|commit|flush|delete|merge(` or `insert|update|delete(`) must be inside a declared shell function (innermost enclosing def).
- The declared pure core must hold at least 50% of the candidate's functions.
- Write sites at most 12 (the original's count); intra-set call edges at most 66 (25% below the original's 89); physical lines after canonical formatting (ruff format 0.17.3, line length 100) at most 1,345 (10% below 1,495); functions at most 55 (20% below 69). Statements are reported, not targeted.
- All seven files stay at their module paths; a file may become thin but may not be removed or renamed, because tests import and patch these module paths. Names that tests patch or override (kind "patched" in the surface: `_cited_baseline`, `_lifecycle_store`, `_pod_of`, `configured_pods`, `read_parameters` of lifecycle_group; `_consumed`, `baseline_view`, `live_evidence` of lifecycle_evidence; `study_baseline` and `project_root` of lifecycle_root; `LiveEvidence.for_window` of lifecycle_evidence; `LifecycleStore` of lifecycle_store) must remain module globals that the module's own functions look up by name at call time (no capture into closures, defaults, class attributes or import-time aliases in another module). The instance attributes tests set or read (`MonitorRunner._rules`, `_append`, `_evidence`, `_monitored`, `_engine`; `LifecycleStore._engine`, `standing`) must keep their names and roles.
- Keep the statement text and parameters of every SQL statement unchanged; the acceptance arm records them (Level 2) and judges differences. Keep the order and fields of emitted records, the text of CLI output and of error messages, and the check order that decides which refusal is raised: they are Level 1 observations.
- No file outside the seven scope files may change; no test, recorded corpus or apparatus file may be touched; such a candidate is void.
